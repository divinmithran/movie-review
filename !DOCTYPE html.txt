function esc(s){var d=document.createElement("div");d.textContent=s;return d.innerHTML}
function drawChips(){
  $("chips").innerHTML=genres.map(function(g){return '<button type="button" class="chip" aria-pressed="'+(state.g===g)+'" data-g="'+g+'">'+g+'</button>'}).join("");
}
function draw(){
  var list=reviews.map(function(r,i){r.i=i;return r}).filter(function(r){
    return (state.g==="All"||r.g===state.g)&&r.t.toLowerCase().indexOf(state.q.toLowerCase())>-1;
  });
  list.sort(function(a,b){
    if(state.sort==="top")return b.r-a.r||a.t.localeCompare(b.t);
    if(state.sort==="new")return b.i-a.i;
    return a.t.localeCompare(b.t);
  });
  $("count").textContent=list.length+(list.length===1?" review":" reviews");
  $("list").innerHTML=list.length?list.map(function(r){
    return '<article class="ticket"><div class="main"><h2>'+esc(r.t)+'</h2><p class="meta">'+esc(r.g)+(r.y?', '+esc(String(r.y)):'')+'</p><p class="hook">'+esc(r.h)+'</p><details><summary>Read the review</summary><p>'+esc(r.b)+'</p></details></div><div class="stub"><b>'+r.r+'</b><small>out of 5</small></div></article>';
